infix to postfix
----------------
PRECEDENCE = {"+": 1, "-": 1, "*": 2, "/": 2, "**": 3}

operators = Stack()
output = Stack()

for token in expr:

    if token == "(":
        operators.push(token)

    elif token == ")":
        while not operators.isEmpty() and operators.peek() != "(":
            output.push(operators.pop())
        operators.pop()

    elif token.isalnum():
        output.push(token)

    else:
        while not operators.isEmpty() and operators.peek() != "(" and (PRECEDENCE[operators.peek()] > PRECEDENCE[token] or (PRECEDENCE[operators.peek()] == PRECEDENCE[token] and token != "**")):
            output.push(operators.pop())
        operators.push(token)

while not operators.isEmpty():
    output.push(operators.pop())

return output

Infix_to_postfix.py
PRECEDENCE = {"+": 1, "-": 1, "*": 2, "/": 2, "^": 3}

output = ""
operator_stack = Stack()
i = 0

while i < len(expression):
    char = expression[i]

    if char.isdigit():
        number = ""
        while i < len(expression) and expression[i].isdigit():
            number += expression[i]
            i += 1
        output += number + " "
        i -= 1

    elif char == "(":
        operator_stack.push(char)

    elif char == ")":
        while not operator_stack.is_empty() and operator_stack.peek() != "(":
            output += operator_stack.pop() + " "
        operator_stack.pop()

    elif char in PRECEDENCE:
        while not operator_stack.is_empty() and operator_stack.peek() != "(" and PRECEDENCE[operator_stack.peek()] >= PRECEDENCE[char]:
            output += operator_stack.pop() + " "
        operator_stack.push(char)

    elif not char.isspace():
        output += char + " "

    i += 1

while not operator_stack.is_empty():
    output += operator_stack.pop() + " "

return output.strip()
